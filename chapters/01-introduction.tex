\chapter{ابدأ مع كيان}
\label{ch:intro}

\section{ما هو كيان؟}
\label{sec:intro-minute}

\textbf{كيان} يجمع الموظفين والمهام والطلبات والملفات في مساحة واحدة. يعرض لك المسؤول عن كل عمل وحالته، بحسب صلاحياتك.

\begin{table}[H]
\centering
\caption{ماذا تحتاج؟}
\label{tab:intro-minute}
\rcZebra
\begin{tabularx}{\linewidth}{R{3.2cm} Y Y}
\rcHeaderRow
\thh{تحتاج إلى} & \thh{استخدم} & \thh{سترى}\\
متابعة عمل & المهام & مسؤول، موعد، حالة\\
تنظيم الفريق & الموارد البشرية & موظف، وظيفة، دوام\\
جمع طلبات ببيانات مكتملة & النماذج والطلبات & بيانات الطلب ومسار مراجعته\\
العثور على مستند & الملفات والمرفقات & المستند المرتبط بالعمل\\
مراجعة التقدم & المشاريع ولوحات المعلومات & حالة العمل المسجل\\
\end{tabularx}
\end{table}

حدّد مسؤول كل عمل، وأدخل بيانات صحيحة، وحدّث حالته عند إنجازه.

\section{مثال: شركة الأفق}
\label{sec:intro-entity}

مثالنا وهمي: شركة الأفق للخدمات في دمشق وحلب. سارة مسؤولة الموارد البشرية، وعمر منسق العمليات.

\section{أول دخول}
\label{sec:intro-app-map}

\textbf{جهّز:} عنوان المؤسسة وطريقة الدخول من المسؤول. مثال العنوان: \en{hzn.kayanos.app}. لإنشاء مؤسسة، راجع الفصل~\ref{ch:org-setup}.

\begin{enumerate}
  \item افتح العنوان، واختر البريد أو الهاتف أو رقم الموظف حسب إعداد حسابك، ثم سجّل الدخول.
  \item تحقق من اسم المؤسسة قبل إدخال البيانات.
\end{enumerate}
